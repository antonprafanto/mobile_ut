# UI Components

Ionic menyediakan library komponen UI yang kaya dan siap pakai. Mari explore komponen-komponen penting dan cara menggunakannya.

## Button Components

### Basic Buttons

```html

<!-- Button dengan color -->

<ion-button color="warning">Warning</ion-button>

<!-- Button fills -->

<ion-button fill="outline">Outline</ion-button>
<ion-button fill="clear">Clear</ion-button>

<!-- Button sizes -->
<ion-button size="small">Small</ion-button>
<ion-button size="default">Default</ion-button>
<ion-button size="large">Large</ion-button>

<!-- Full width -->
<ion-button expand="full">Full Width</ion-button>
<ion-button expand="block">Block</ion-button>

<!-- With icons -->
<ion-button>
  <ion-icon slot="start" name="star"></ion-icon>
  Favorit
</ion-button>

<ion-button>
  Submit
  <ion-icon slot="end" name="arrow-forward"></ion-icon>
</ion-button>

<!-- Icon only -->
<ion-button>
  <ion-icon slot="icon-only" name="heart"></ion-icon>
</ion-button>
```

## Loading & Spinner

```typescript
// loading.ts
import { LoadingController } from '@ionic/angular/standalone';

export class MyPage {
  constructor(private loadingCtrl: LoadingController) {}

  async showLoading() {
    const loading = await this.loadingCtrl.create({
      message: 'Please wait...',
      duration: 3000,
      spinner: 'circles'
    });

    await loading.present();
  }
}
```

## Alert Dialogs

```typescript
import { AlertController } from '@ionic/angular/standalone';

export class MyPage {
  constructor(private alertCtrl: AlertController) {}

  async showAlert() {
    const alert = await this.alertCtrl.create({
      header: 'Confirm',
      message: 'Are you sure?',
      buttons: [
        {
          text: 'Cancel',
          role: 'cancel'
        },
        {
          text: 'OK',
          handler: () => {
            console.log('Confirmed!');
          }
        }
      ]
    });

    await alert.present();
  }

  async showPrompt() {
    const alert = await this.alertCtrl.create({
      header: 'Enter Name',
      inputs: [
        {
          name: 'username',
          type: 'text',
          placeholder: 'Username'
        }
      ],
      buttons: [
        'Cancel',
        {
          text: 'Save',
          handler: (data) => {
            console.log('Username:', data.username);
          }
        }
      ]
    });

    await alert.present();
  }
}
```

## Practical Example: Product List

```typescript
// products.page.ts
export class ProductsPage {
  products = [
    {
      id: 1,
      name: 'Laptop',
      price: 15000000,
      image: 'assets/laptop.jpg',
      stock: 5,
      rating: 4.5
    },
    {
      id: 2,
      name: 'Mouse',
      price: 250000,
      image: 'assets/mouse.jpg',
      stock: 15,
      rating: 4.0
    }
  ];

  constructor(
    private toastCtrl: ToastController,
    private alertCtrl: AlertController
  ) {}

  async addToCart(product: any) {
    const toast = await this.toastCtrl.create({
      message: `${product.name} added to cart`,
      duration: 2000,
      color: 'success',
      icon: 'cart'
    });
    await toast.present();
  }

  async deleteProduct(product: any) {
    const alert = await this.alertCtrl.create({
      header: 'Confirm Delete',
      message: `Delete ${product.name}?`,
      buttons: [
        'Cancel',
        {
          text: 'Delete',
          role: 'destructive',
          handler: () => {
            // Delete logic
          }
        }
      ]
    });
    await alert.present();
  }
}
```

## Best Practices

1. **Use Ionic Components** - Jangan recreate yang sudah ada
2. **Consistent Spacing** - Gunakan padding utilities: `ion-padding`, `ion-margin`
3. **Responsive Design** - Test di berbagai screen sizes
4. **Dark Mode** - Support dengan CSS variables
5. **Accessible** - Gunakan proper labels dan ARIA attributes
